def get_entity_location(team, entity_id):
    if entity_id == BLUE_FLAG:
        z = team.protocol.map.get_z(220, 250)
        return (220, 250, z)
    if entity_id == BLUE_BASE:
        z = team.protocol.map.get_z(119, 250)
        return (119, 250, z)
    if entity_id == GREEN_BASE:
        z = team.protocol.map.get_z(395, 250)
        return (395, 250, z)
    if entity_id == GREEN_FLAG:
        z = team.protocol.map.get_z(289, 250)
        return (289, 250, z)
