def get_spawn_location(connection):

    if connection.team is connection.protocol.team_1:
        x = random.randrange(79, 89)
        y = random.randrange(235, 277)
        z = connection.protocol.map.get_z(x,y)
        return (x, y, z)
    if connection.team is connection.protocol.team_2:
        x = random.randrange(425, 435)
        y = random.randrange(235, 277)
        z = connection.protocol.map.get_z(x,y)
        return (x, y, z)
