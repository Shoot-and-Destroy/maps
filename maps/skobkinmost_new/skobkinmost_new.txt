from pyspades.constants import *
import random

def get_entity_location(team, entity_id):

    if entity_id == BLUE_FLAG:
        x = random.randrange(311, 351)
        y = random.randrange(40, 70)
        z = team.protocol.map.get_z(x, y)
        return (x, y, z)

    if entity_id == BLUE_BASE:
        x = random.randrange(452, 507)
        y = random.randrange(115, 225)
        z = team.protocol.map.get_z(x, y)
        return (x, y, z)

    if entity_id == GREEN_FLAG:
        x = random.randrange(156, 186)
        y = random.randrange(340, 390)
        z = team.protocol.map.get_z(x, y)
        return (x, y, z)

    if entity_id == GREEN_BASE:
        x = random.randrange(270, 342)
        y = random.randrange(457, 509)
        z = team.protocol.map.get_z(x, y)
        return (x, y, z)
