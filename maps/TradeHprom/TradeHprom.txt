def get_spawn_location(connection):

    if connection.team is connection.protocol.team_1:
        where=int(random.randrange(1,3))
        if where == 1:        
            x = random.randrange(134, 154)
            y = random.randrange(342, 362)
            z = connection.protocol.map.get_z(x,y)
            return (x, y, z)
        if where == 2:
            x = random.randrange(134, 154)
            y = random.randrange(145, 169)
            z = connection.protocol.map.get_z(x,y)
            return (x, y, z)
    if connection.team is connection.protocol.team_2:
        where=int(random.randrange(1,3))
        if where == 1:
            x = random.randrange(357, 378)
            y = random.randrange(342, 362)
            z = connection.protocol.map.get_z(x,y)
            return (x, y, z)
        if where == 2:
            x = random.randrange(358, 378)
            y = random.randrange(148, 168)
            z = connection.protocol.map.get_z(x,y)
            return (x, y, z)
