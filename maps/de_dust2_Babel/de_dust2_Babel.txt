def get_spawn_location(connection):
    # --- КОМАНДА 1 (Синие) ---
    if connection.team is connection.protocol.team_1:
        # Выбираем одну из двух зон с шансом 50%
        zone = random.randint(1, 2)
        
        if zone == 1:
            # X: 196-211, Y: 261-278
            x = random.randrange(196, 212)
            y = random.randrange(273, 288)
        else:
            # X: 203-212, Y: 210-227
            x = random.randrange(203, 213)
            y = random.randrange(210, 228)

    # --- КОМАНДА 2 (Зеленые) ---
    elif connection.team is connection.protocol.team_2:
        # X: 320-326, Y: 234-323
        x = random.randrange(320, 327)
        y = random.randrange(234, 324)
        
    # --- НАБЛЮДАТЕЛИ ---
    else:
        return (256, 256, 60)

    # Получаем высоту земли в этой точке
    z = connection.protocol.map.get_z(x, y)
    return (x, y, z)

def get_entity_location(team, entity_id):
    # --- ПАЛАТКИ (БАЗЫ) ---
    if entity_id == BLUE_BASE:
        x = 193
        y = 218
    elif entity_id == GREEN_BASE:
        x = 323
        y = 295
    else:
        return (0, 0, 0)
    
    # Получаем высоту для палатки
    z = team.protocol.map.get_z(x, y)
    return (x, y, z)
