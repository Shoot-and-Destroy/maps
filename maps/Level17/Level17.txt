from random import randint

name        = 'Level 17'
version     = '1.0'
description = 'Catacombs - a trench maze cut into rock, lit by a fresco that is no map'
fog         = (46, 38, 30)
SPAWN       = (205, 377)

extensions = {
	'base_locations': [
		(111, 389),
		(331, 228),
		(419, 321),
		(244, 177),
		(109, 278),
		(135, 116),
		(220, 419),
		(215, 294)
	],
    'players_spawn_start' : SPAWN
}

def get_spawn_location(player):
	x1, y1 = SPAWN

	x = randint(x1, x1 + 9)
	y = randint(y1, y1 + 9)

	return (x, y, player.protocol.map.get_z(x, y))
