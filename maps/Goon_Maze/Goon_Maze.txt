def get_entity_location(team, entity_id):
    if entity_id == BLUE_FLAG:
        z = team.protocol.map.get_z(192, 224)
        return (192, 224, z)
    if entity_id == GREEN_FLAG:
        z = team.protocol.map.get_z(320, 296)
        return (320, 296, z)
