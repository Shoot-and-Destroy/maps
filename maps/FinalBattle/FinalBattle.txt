# --- SPAWN LOGIC ---
def get_spawn_location(connection):
    if connection.team is connection.protocol.team_1:
        x = random.randrange(176, 204)
        y = random.randrange(240, 271)
        z = connection.protocol.map.get_z(x,y)
        return (x, y, z)
    if connection.team is connection.protocol.team_2:
        x = random.randrange(307, 336)
        y = random.randrange(240, 271)
        z = connection.protocol.map.get_z(x,y)
        return (x, y, z)

def get_entity_location(team, entity_id):
    if entity_id == BLUE_BASE:
        return (253, 211, team.protocol.map.get_z(253, 211))
    if entity_id == GREEN_BASE:
        return (259, 300, team.protocol.map.get_z(259, 300))
    return (0,0,0)
