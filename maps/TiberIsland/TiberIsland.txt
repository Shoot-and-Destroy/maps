def get_spawn_location(connection):
    # --- КОМАНДА 1 (Синие) ---
    if connection.team is connection.protocol.team_1:
        # X: 169-174, Y: 238-277
        x = random.randrange(169, 174)
        y = random.randrange(239, 277)
        
    # --- КОМАНДА 2 (Зеленые) ---
    elif connection.team is connection.protocol.team_2:
        # X: 350-356, Y: 243-280
        x = random.randrange(350, 357)
        y = random.randrange(243, 280)
        
    # Наблюдатели
    else:
        return (256, 256, 60)

    # Получаем высоту земли
    z = connection.protocol.map.get_z(x, y)
    return (x, y, z)

def get_entity_location(team, entity_id):
    # --- ПАЛАТКИ (БАЗЫ) ---
    if entity_id == BLUE_BASE:
        x = 143
        y = 215
    elif entity_id == GREEN_BASE:
        x = 394
        y = 256
    else:
        # Для других сущностей (например, intel)
        return (0, 0, 0)
    
    # Получаем высоту для палатки
    z = team.protocol.map.get_z(x, y)
    return (x, y, z)
