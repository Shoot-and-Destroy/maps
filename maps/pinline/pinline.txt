def get_spawn_location(connection):
	if connection.team is connection.protocol.blue_team:
		x=70
		y=256
		return (x,y,connection.protocol.map.get_z(x,y)-1)
	elif connection.team is connection.protocol.green_team:
		x=440
		y=256
		return (x,y,connection.protocol.map.get_z(x,y)-1)
