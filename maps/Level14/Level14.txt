from random import randint

name        = 'Level 14'
version     = '1.0'
author      = 'Rakete'
description = 'A hexagonal maze map'
fog         = (51, 161, 201)
SPAWN       = (256, 251)

extensions = {
	'base_locations': [
		(97, 170),
		(261, 76),
        (425, 170),
        (425, 360),
		(261, 455),
		(97, 360)
	],
    'players_spawn_start' : SPAWN
}

def get_spawn_location(player):
	x1, y1 = SPAWN

	x = randint(x1, x1 + 9)
	y = randint(y1, y1 + 9)

	return (x, y, player.protocol.map.get_z(x, y))
