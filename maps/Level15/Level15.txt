from random import randint

name        = 'Level 15'
version     = '1.0'
author      = 'Rakete, (unknown)'
description = 'Wavewall. Found this on aos.party and found it to be good'
SPAWN       = (249, 253)

extensions = {
	'base_locations': [
		(63, 63), (255, 63), (447, 63), (63, 255),(447, 255), (63, 447), (255, 447), (447, 447)
	],
    'players_spawn_start' : SPAWN
}

def get_spawn_location(player):
	x1, y1 = SPAWN

	x = randint(x1, x1 + 9)
	y = randint(y1, y1 + 9)

	return (x, y, player.protocol.map.get_z(x, y))
