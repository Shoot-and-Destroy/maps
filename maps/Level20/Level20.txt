from random import randint

name        = 'Level 20'
version     = '1.0'
description = 'Mandelbrot Coast - canyons cut on one fractal, painted with another'
fog         = (24, 18, 52)
SPAWN       = (251, 251)

extensions = {
	'base_locations': [
		(247, 118),
		(427, 77),
		(120, 390),
		(267, 280),
		(258, 397),
		(444, 314),
		(334, 184),
		(97, 102)
	],
    'players_spawn_start' : SPAWN
}

def get_spawn_location(player):
	x1, y1 = SPAWN

	x = randint(x1, x1 + 9)
	y = randint(y1, y1 + 9)

	return (x, y, player.protocol.map.get_z(x, y))
