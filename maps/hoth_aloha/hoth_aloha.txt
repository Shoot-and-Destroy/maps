def get_entity_location(team, entity_id):

	if entity_id == BLUE_FLAG or entity_id == GREEN_FLAG:
		return (255,255,0)
		
	if entity_id == BLUE_BASE:
		return (130,255,60)

	if entity_id == GREEN_BASE:
		return (382,255,60)

def get_spawn_location(connection):

	if connection.team is connection.protocol.blue_team:
		x = random.randrange(122,137)
		y = random.randrange(227,287)
		z = 56
		return (x, y, z)

	if connection.team is connection.protocol.green_team:
		x = random.randrange(381,396)
		y = random.randrange(224,284)
		z = 55
		return (x, y, z)
