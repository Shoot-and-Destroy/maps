from random import randint

name        = 'Level 19'
version     = '1.0'
description = 'Karesansui - the sand is raked one way and built another'
fog         = (206, 208, 196)
SPAWN       = (212, 251)

extensions = {
	'base_locations': [
		(291, 149),
		(317, 251),
		(405, 305),
		(129, 71),
		(223, 288),
		(406, 205),
		(422, 64),
		(209, 133)
	],
    'players_spawn_start' : SPAWN
}

def get_spawn_location(player):
	x1, y1 = SPAWN

	x = randint(x1, x1 + 9)
	y = randint(y1, y1 + 9)

	return (x, y, player.protocol.map.get_z(x, y))
