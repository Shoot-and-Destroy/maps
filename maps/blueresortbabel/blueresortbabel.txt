import random
def get_spawn_location(connection):
	br1 = random.choice(BLUE_RECT)
	gr1 = random.choice(GREEN_RECT)
	if connection.team is connection.protocol.blue_team:
		return connection.protocol.get_random_location(True, br1)
	elif connection.team is connection.protocol.green_team:
		return connection.protocol.get_random_location(True, gr1)
