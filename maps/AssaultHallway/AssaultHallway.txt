def get_entity_location(team, entity_id):

    if entity_id == BLUE_FLAG:
        x = random.randrange(79, 104)
        y = random.randrange(238, 272)
        z = team.protocol.map.get_z(x,y)
        return (x,y,z)

    if entity_id == BLUE_BASE:
        x = 91.5
        y = 255
        z = team.protocol.map.get_z(x,y)
        return (x, y, z)

    if entity_id == GREEN_FLAG:
        x = random.randrange(407, 432)
        y = random.randrange(239, 273)
        z = team.protocol.map.get_z(x,y)
        return (x, y, z)

    if entity_id == GREEN_BASE:
        x = 419.5
        y = 255
        z = team.protocol.map.get_z(x,y)
        return (x, y, z)
